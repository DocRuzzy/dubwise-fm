# The dub tape echo (P0.3). A short note from MIDI (the bass) sent to the echo. At 75 BPM
# the default time, a dotted eighth, is 0.75 x 800 ms = 600 ms: 6615 line samples (11,025 a second), less one for the
# converters. Run by tests/run_tests.sh, then the #wav checks at the end.
wait 600
listen
expect echo_div 5
expect echo_t 6614
param ch4.echo 127
param echo.fb 30
# one burst: its first repeat 600 ms after it, then falling (T60 0.8 s)
wav echo-burst.wav
midi 90 3C 64
wait 60
midi 80 3C 00
wait 570
expect echo_ret_db >-40
wait 1700
expect echo_ret_db <-40
wavstop
# the knobs: K1 TIME a division at a time (the slew bends it there), K2 FB, K3 the return
turn K1 1
wait 30
said TIME 1/4
wait 1000
expect echo_div 6
expect echo_t >8790
turn K1 -1
turn K2 2
wait 30
expect ui.echo.fb 32
turn K3 -5
wait 30
expect ui.echo.ret 90
