# Dubwise FM in the host simulator: tests/run_tests.sh, or ./build.sh --host, then
#   app/build/host/dubwise_host app/tests/basic.txt app/build/host
# The platform's basics: MIDI in plays the test voice, settings persist, the screen reader, MASTER, velocity.
wait 600
shot boot.png
expect ble 0
# MIDI in plays the test voice on the bass channel (until the voices of P0.5)
midi 90 3C 64
wait 30
expect note 60
expect midi_out 0
wav note.wav
wait 200
wavstop
expect peak_db_min -30
midi 80 3C 00
wait 50
expect note -1
# PRESET: the test voice's wave; OCT+: quantise; HOME + SELECT: tempo; SAVE keeps them over a restart
turn PRESET 1
wait 30
expect wave 1
expect dirty 1
tap OCT+
wait 30
expect ui.quant 2
press HOME
spin SELECT 3
wait 30
release HOME
wait 30
expect ui.tempo 780
tap SAVE
wait 30
expect dirty 0
expect store_writes 1
reboot
wait 600
expect wave 1
expect ui.quant 2
expect eng.tempo 780
# the screen reader hears the page, then each change
listen
wait 100
said DUBWISE. 78.0 BPM. STOPPED. QUANTISE 2 BARS
tap ARP
wait 30
said BUTTON ARP
tap REC
wait 30
said BUTTON REC
tap OCT-
wait 30
said QUANTISE 1 BAR
# MASTER turned while a note plays: the gain glides, so no sample jumps further than the wave itself does
turn PRESET 1
wait 30
expect wave 0
midi 90 45 64
wait 100
wav master.wav
master 4096
wait 20
master 300
wait 20
master 4096
wait 50
wavstop
midi 80 45 00
# velocity (lib/touch.c): a key landing at 40 reads about 40 (a throw key: velocity-scaled throws are P2.9)
key w12 down 40
wait 60
expect vel >30
expect vel <50
key w12 up
wait 50
# each port's messages counted (plat_midi_count): this is the fifth from USB in this script
midi USB 90 40 50
wait 20
expect midi_in_usb 5
midi USB 80 40 00
