# The platform's basics: MIDI in plays the test voice, settings persist, the screen reader, MASTER, velocity.
wait 600

# MIDI in plays the bass (any channel but 10)

wait 50
expect note -1
# EDIT: the SOUND view of the focused channel (the kick): MAIN (the macros, 5.3), the raw banks, DICE (5.6); OCT+:
# quantise; HOME + SELECT: tempo; SAVE keeps them
tap EDIT
tap EDIT
wait 30
expect sound_bank 2
turn K1 3
wait 30
expect ui.v1.tune 45
expect dirty 1
tap EDIT
tap EDIT
tap EDIT
wait 30
expect sound_bank 0
tap OCT+
wait 30
expect ui.quant 2
press HOME
spin SELECT 3
wait 30
release HOME
wait 30
expect ui.tempo 780
tap SAVE
wait 30
expect dirty 0
expect store_writes 7
reboot
wait 600
expect ui.v1.tune 45
expect ui.quant 2
expect eng.tempo 780
# the screen reader hears the page, then each change
listen
